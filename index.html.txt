<script>
async function loadData() {
    const res = await fetch("/earthquakes");
    const data = await res.json();

    let rows = "";

    data.features.slice(0, 10).forEach(eq => {
        const p = eq.properties;

        rows += `
        <tr>
            <td>${p.mag}</td>
            <td>${p.place}</td>
            <td>${new Date(p.time).toLocaleString()}</td>
        </tr>`;
    });

    document.getElementById("data").innerHTML = rows;
}
</script>
